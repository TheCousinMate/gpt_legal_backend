def extraer_norma_desde_html(html: str) -> dict:
    if "<title>" in html:
        start = html.find("<title>") + len("<title>")
        end = html.find("</title>")
        titulo = html[start:end]
    else:
        titulo = "Norma no encontrada"
    return {"titulo": titulo}

def clasificar_consulta(consulta: str) -> str:
    if not consulta:
        return "No clasificada"
    if "civil" in consulta.lower():
        return "Derecho Civil"
    elif "penal" in consulta.lower():
        return "Derecho Penal"
    return "General"

from fastapi import APIRouter
from modules.root import router as root_router
from modules.reflexion import reflexionar
from modules.predict import predecir_riesgo_legal
from modules.vision_forense import analizar_imagen
from modules.forensics_proxy import buscar_beneficiarios
from modules.legislacion_bcn import buscar_norma_por_id
from modules.bcn_text_search_parser import buscar_normas_por_texto
from modules.prediccion_bayesiana import modelo_bayesiano
from fastapi import UploadFile, File

router = APIRouter()
router.include_router(root_router)

@router.post("/reflexion/")
def endpoint_reflexion(payload: dict):
    return reflexionar(payload.get("mensaje", ""))

@router.post("/predict")
def endpoint_predict(payload: dict):
    return predecir_riesgo_legal(payload.get("consulta", ""))

@router.post("/api/prediccion/")
def endpoint_prediccion(payload: dict):
    return modelo_bayesiano(payload.get("texto", ""))

@router.get("/api/bcn/norma/{norma_id}")
def endpoint_bcn_norma(norma_id: int):
    return buscar_norma_por_id(norma_id)

@router.get("/legislacion/buscar")
def endpoint_buscar_legislacion(palabra: str = ""):
    return buscar_normas_por_texto(palabra)

@router.post("/forensics/search")
def endpoint_forensics(payload: dict):
    return buscar_beneficiarios(payload.get("empresa", ""))

@router.post("/vision/")
async def endpoint_vision(file: UploadFile = File(...)):
    content = await file.read()
    return analizar_imagen(content)
