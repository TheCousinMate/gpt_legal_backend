import os
import sys
import logging

from pathlib import Path

# Configurar logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.StreamHandler(sys.stdout)
    ]
)

logger = logging.getLogger(__name__)

def setup_environment():
    """Configura el entorno de ejecución"""
    
    # Verificar archivos críticos
    critical_files = [
        "main.py",
        "utils_output.py",
        "requirements.txt"
    ]
    
    missing_files = []
    for file in critical_files:
        if not Path(file).exists():
            missing_files.append(file)
    
    if missing_files:
        logger.error(f"Archivos críticos faltantes: {missing_files}")
        sys.exit(1)
    
    # Verificar directorio modules
    if not Path("modules").exists():
        logger.error("Directorio 'modules' no encontrado")
        sys.exit(1)
    
    # Crear directorios si no existen
    Path("static").mkdir(exist_ok=True)
    Path("logs").mkdir(exist_ok=True)
    
    logger.info("Entorno configurado correctamente")

def get_port():
    """Obtiene el puerto desde variables de entorno"""
    port = int(os.environ.get("PORT", 10000))
    logger.info(f"Puerto configurado: {port}")
    return port

def get_host():
    """Obtiene el host para bind"""
    # En Render siempre usar 0.0.0.0
    host = "0.0.0.0"
    logger.info(f"Host configurado: {host}")
    return host

def main():
    """Función principal de lanzamiento"""
    logger.info("🚀 Iniciando Forensic Legal Cognition 4.1")
    
    try:
        # Configurar entorno
        setup_environment()
        
        # Obtener configuración
        host = get_host()
        port = get_port()
        
        # Configuración de uvicorn para producción
        config = {
            "app": "main:app",
            "host": host,
            "port": port,
            "workers": 1,  # Single worker en Render
            "loop": "auto",
            "http": "auto",
            "log_level": "info",
            "access_log": True,
            "use_colors": False,  # Mejor para logs en Render
            "reload": False,  # Nunca en producción
        }
        
        logger.info(f"Configuración uvicorn: {config}")
        logger.info("✅ Backend iniciando...")
        
        # Iniciar servidor
        uvicorn.run(**config)
        
    except KeyboardInterrupt:
        logger.info("🛑 Servidor detenido por usuario")
    except Exception as e:
        logger.error(f"❌ Error crítico al iniciar servidor: {e}")
        sys.exit(1)

if __name__ == "__main__":
    main()

    documento_analizado: str
    criterios: list[str]

def reflexionar(data: ReflexionInput):
    if not data.documento_analizado.strip():

    if not data.criterios:

        "justificacion": "Evaluación doctrinal automatizada",
        "logs": {"input": data.dict()}
    }
    return JSONResponse(content=output_dual(resultado))

"""
Script de testing completo para Forensic Legal Cognition API 4.1
"""

import json
import time
import argparse
from typing import Optional, Dict, Any

try:
    import httpx
except ModuleNotFoundError:
    print("❌ ERROR: No se pudo importar 'httpx'. Usa:")
    print("   pip install httpx")
    exit(1)

try:
    import ssl
except ModuleNotFoundError:
    print("❌ ERROR: No se puede importar 'ssl'.")
    exit(1)

BASE_URL = "https://gpt-legal-backend.onrender.com"
LOCAL_URL = "http://127.0.0.1:8000"
TIMEOUT = 30.0

class APITester:
    def __init__(self, base_url: str = BASE_URL):
        self.base_url = base_url
        self.client = httpx.Client(timeout=TIMEOUT)
        self.results = []

    def log_result(self, test_name: str, success: bool, response_data: Any = None, error: str = None):
        result = {
            "test": test_name,
            "success": success,
            "timestamp": time.strftime("%H:%M:%S"),
            "response_preview": str(response_data)[:200] if response_data else None,
            "error": error
        }
        self.results.append(result)
        status = "✅" if success else "❌"
        print(f"{status} {test_name}")
        if error:
            print(f"   Error: {error}")
        if response_data and success:
            print(f"   Response: {str(response_data)[:100]}...")
        print()

    def safe_request(self, method: str, endpoint: str, **kwargs) -> Optional[httpx.Response]:
        url = f"{self.base_url}{endpoint}"
        try:
            return self.client.request(method, url, **kwargs)
        except Exception as e:
            print(f"❌ Request failed: {e}")
            return None

    def test_health_check(self):
        response = self.safe_request("GET", "/")
        if response and response.status_code == 200:
            self.log_result("Health Check", True, response.json())
            return True
        else:
            self.log_result("Health Check", False, error=f"Status: {response.status_code if response else 'No response'}")
            return False

    def test_status_json(self):
        response = self.safe_request("GET", "/status/json")
        if response and response.status_code == 200:
            self.log_result("Status JSON", True, response.json())
            return True
        else:
            self.log_result("Status JSON", False, error=f"Status: {response.status_code if response else 'No response'}")
            return False

    def test_error_handling(self):
        error_tests = [
            ("POST", "/predict", {"invalid": "data"}),
            ("POST", "/reflexion/", {"documento_analizado": ""}),
            ("GET", "/legislacion/buscar", {"palabra": "a"}),
            ("GET", "/legislacion/norma/999999999", None),
            ("POST", "/api/prediccion/", {"contexto": "muy corto"}),
        ]
        handled_errors = 0
        for method, endpoint, data in error_tests:
            if method == "GET" and isinstance(data, dict):
                response = self.safe_request(method, endpoint, params=data)
            elif method == "GET":
                response = self.safe_request(method, endpoint)
            else:
                response = self.safe_request(method, endpoint, json=data)

            test_name = f"Error Handling {method} {endpoint}"
            if response and response.status_code >= 400:
                try:
                    _ = response.json()
                    self.log_result(test_name, True, f"Error handled: {response.status_code}")
                    handled_errors += 1
                except:
                    self.log_result(test_name, True, f"Error handled: {response.status_code} (no JSON)")
                    handled_errors += 1
            else:
                self.log_result(test_name, False, error="Error not properly handled")
        return handled_errors >= len(error_tests) // 2

    def run_comprehensive_test(self):
        print("🚀 Iniciando tests comprehensivos de Forensic Legal Cognition API 4.1")
        print("=" * 70)
        start_time = time.time()
        results = [
            self.test_health_check(),
            self.test_status_json(),
            self.test_error_handling()
        ]
        total_time = time.time() - start_time
        passed = sum(results)
        total = len(results)
        print("=" * 70)
        print(f"✅ Tests exitosos: {passed}/{total} ({(passed/total)*100:.1f}%)")
        print(f"⏱️  Tiempo total: {total_time:.2f} segundos")
        print("=" * 70)
        return passed == total

def main():

import json
from typing import Dict, Any
from datetime import datetime

def output_dual(data: Dict[str, Any]) -> Dict[str, str]:
    defaults = {
        'respuesta': 'Sin respuesta generada',
        'clasificacion': 'No clasificado',
        'justificacion': 'Sin justificación',
        'base_legal': {},
        'derecho_comparado': {},
        'contradicciones': [],
        'estandares_internacionales': [],
        'esquema_visual': '',
        'sintesis_audio': '',
        'fuentes': [],
        'logs': {},
        'timestamp': datetime.now().isoformat(),
        'version': '4.1.0'
    }
    complete_data = {**defaults, **data}
    output_json = json.dumps(complete_data, ensure_ascii=False, indent=2)
    output_md = f"""# 📋 Análisis Jurídico Automatizado

## 🎯 Respuesta Principal
{complete_data['respuesta']}

---

## 📊 Clasificación del Caso
**Nivel:** {complete_data['clasificacion']}  
**Justificación:** {complete_data['justificacion']}

---

## ⚖️ Marco Legal Aplicable
{format_legal_framework(complete_data['base_legal'])}

---

## 🌍 Derecho Comparado
{format_comparative_law(complete_data['derecho_comparado'])}

---

## ⚠️ Contradicciones Detectadas
{format_contradictions(complete_data['contradicciones'])}

---

## 🏛️ Estándares Internacionales
{format_international_standards(complete_data['estandares_internacionales'])}

---

## 📈 Esquema Visual
{complete_data['esquema_visual'] or 'No disponible'}

---

## 🎵 Síntesis para Audio
{complete_data['sintesis_audio'] or 'No disponible'}

---

## 📚 Fuentes Consultadas
{format_sources(complete_data['fuentes'])}

---

## 🔍 Información Técnica
**Timestamp:** {complete_data['timestamp']}  
**Versión:** {complete_data['version']}  
**Logs:** ```json
{json.dumps(complete_data['logs'], ensure_ascii=False, indent=2)}
```"""
    return {"json": output_json, "markdown": output_md.strip()}

def format_legal_framework(base_legal: Dict[str, str]) -> str:
    return "\n".join(f"- **{k}:** {v}" for k, v in base_legal.items()) or "*No se identificó marco legal específico.*"

def format_comparative_law(dc: Dict[str, str]) -> str:
    return "\n".join(f"- **{k}:** {v}" for k, v in dc.items()) or "*No se aplicó análisis de derecho comparado.*"

def format_contradictions(lista: list) -> str:
    return "\n".join(f"{i+1}. {item}" for i, item in enumerate(lista)) or "*No se detectaron contradicciones normativas.*"

def format_international_standards(estandares: list) -> str:
    return "\n".join(f"- {s}" for s in estandares) or "*No se aplicaron estándares internacionales.*"

def format_sources(fuentes: list) -> str:
    formatted = []
    for i, fuente in enumerate(fuentes, 1):
        if isinstance(fuente, str):
            formatted.append(f"{i}. {fuente}")
        elif isinstance(fuente, dict):
            formatted.append(f"{i}. [{fuente.get('title', 'Sin título')}]({fuente.get('url', '#')})")
        else:
            formatted.append(f"{i}. {str(fuente)}")
    return "\n".join(formatted) or "*No se registraron fuentes específicas.*"

def create_error_response(msg: str, details: str = "") -> Dict[str, str]:
    return output_dual({
        'respuesta': f"Error: {msg}",
        'clasificacion': 'Error',
        'justificacion': details or "Se produjo un error durante el procesamiento",
        'logs': {'error': msg, 'details': details, 'timestamp': datetime.now().isoformat()}
    })

def create_success_response(message: str, data: Dict[str, Any] = None) -> Dict[str, str]:
    return output_dual({

        'clasificacion': 'Exitoso',
        'justificacion': 'Operación completada correctamente',
        'logs': {'status': 'success', 'data': data or {}, 'timestamp': datetime.now().isoformat()}
    })

import requests
from bs4 import BeautifulSoup
from fastapi import APIRouter
from fastapi.responses import JSONResponse

router = APIRouter()

@router.get("/api/bcn/norma/{id_norma}")
def norma_bcn(id_norma: int):
    """
    Extrae información de una norma chilena desde el sitio de la Biblioteca del Congreso Nacional (BCN).
    """
    try:
        url = f"https://www.bcn.cl/leychile/Navegar?idNorma={id_norma}"
        headers = {"User-Agent": "ForensicLegalBot/1.0"}
        response = requests.get(url, headers=headers, timeout=10)
        response.raise_for_status()
        soup = BeautifulSoup(response.text, "html.parser")

        titulo_tag = soup.find("h1") or soup.find("title")
        titulo = titulo_tag.get_text(strip=True) if titulo_tag else "Título no disponible"

        fecha_tag = soup.find(class_="fecha-publicacion")
        fecha = fecha_tag.get_text(strip=True) if fecha_tag else "Fecha no disponible"

        resumen_tag = soup.find("div", class_="resumen")
        if resumen_tag:
            resumen = resumen_tag.get_text(strip=True)
        else:
            primer_p = soup.find("p")
            resumen = primer_p.get_text(strip=True) if primer_p else "Resumen no disponible"

        resultado = {
            "id": id_norma,
            "titulo": titulo,
            "fecha_publicacion": fecha,
            "contenido": resumen
        }
        return JSONResponse(content=resultado)

    except Exception as e:
        return JSONResponse(status_code=500, content={
            "error": "No se pudo extraer la norma desde BCN.",
            "detalle": str(e)
        })

def interpretar_input(mensaje: str) -> dict:
    mensaje = mensaje.strip()

    if not mensaje or len(mensaje) < 10:
        raise HTTPException(status_code=400, detail="Mensaje demasiado corto para interpretar.")

    try:
        system_prompt = (
            "Eres un modelo jurídico chileno. Dado un mensaje, clasifícalo en una de estas tres categorías:\n"
            "- 'predict' si describe un caso o problema legal concreto\n"
            "- 'reflexion' si es una consulta doctrinal, constitucional o interpretativa\n"
            "- 'vision' si hace referencia a un documento escaneado, imagen o evidencia visual.\n"
            "Responde SOLO con una de estas palabras: predict, reflexion, vision."
        )

        completion = openai.ChatCompletion.create(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": mensaje}
            ],
            temperature=0,
            max_tokens=5
        )

        decision = completion.choices[0].message["content"].strip().lower()

        if decision == "predict":
            return {
                "endpoint": "/predict",
                "input_json": {"contexto": mensaje},
                "interpretacion": "Clasificado como análisis de riesgo jurídico concreto."
            }

        elif decision == "reflexion":
            return {
                "endpoint": "/reflexion",
                "input_json": {"consulta": mensaje},
                "interpretacion": "Clasificado como reflexión o consulta doctrinal."
            }

        elif decision == "vision":
            return {
                "endpoint": "/vision",
                "input_json": {"mensaje": mensaje},
                "interpretacion": "Clasificado como contenido visual o documento escaneado."
            }

        else:
            raise ValueError("Respuesta IA fuera de rango")

    except Exception as e:
        mensaje_lower = mensaje.lower()
        if any(p in mensaje_lower for p in ["despido", "finiquito", "contrato", "demanda", "juicio", "causa"]):

                "endpoint": "/predict",
                "input_json": {"contexto": mensaje},

            }

            return {
                "endpoint": "/reflexion",
                "input_json": {"consulta": mensaje},
                "interpretacion": "Fallback: clasificación básica por palabra clave (doctrina)."
            }
        elif any(p in mensaje_lower for p in ["imagen", "documento", "foto", "escaneado"]):
            return {
                "endpoint": "/vision",
                "input_json": {"mensaje": mensaje},
                "interpretacion": "Fallback: clasificación básica por palabra clave (visual)."
            }
        raise HTTPException(status_code=500, detail=f"Error interpretando mensaje: {str(e)}")

import os
import re

router = APIRouter()

# Cargar modelo entrenado
modelo_path = os.path.join("modules", "modelo_bayesiano.joblib")
if not os.path.isfile(modelo_path):
    raise FileNotFoundError(f"Modelo no encontrado en {modelo_path}")
modelo = joblib.load(modelo_path)

# Clase de entrada
class TextoEntrada(BaseModel):
    texto: str

# Preprocesamiento mínimo
stopwords = {"de", "la", "que", "el", "en", "y", "a", "los", "del", "se"}  # puedes ampliar

def limpiar_texto(texto: str) -> str:
    texto = texto.lower()
    texto = re.sub(r"[^a-zñáéíóú\s]", "", texto)
    palabras = texto.split()
    palabras_filtradas = [p for p in palabras if p not in stopwords]
    return " ".join(palabras_filtradas)

# Ruta del API
@router.post("/prediccion-bayesiana", tags=["Predicción Bayesiana"])
def predecir(texto_entrada: TextoEntrada):
    if not texto_entrada.texto.strip():
        raise HTTPException(status_code=400, detail="El texto no puede estar vacío.")

    texto_limpio = limpiar_texto(texto_entrada.texto)
    prediccion = modelo.predict([texto_limpio])[0]
    probas = modelo.predict_proba([texto_limpio])[0]
    etiquetas = modelo.classes_

    return {
        "resultado": prediccion,
        "probabilidades": dict(zip(etiquetas, probas))
    }

import logging

def setup_logger(name: str):
    logger = logging.getLogger(name)
    if not logger.handlers:
        logger.setLevel(logging.INFO)
        ch = logging.StreamHandler()
        formatter = logging.Formatter('[%(asctime)s] %(levelname)s - %(message)s')
        ch.setFormatter(formatter)
        logger.addHandler(ch)
    return logger

from fastapi import APIRouter, HTTPException, UploadFile, File, Body
from fastapi.responses import JSONResponse
import pytesseract
from PIL import Image
import io
import base64

# Ruta explícita para Windows (ajusta si usas Linux/Render)
pytesseract.pytesseract.tesseract_cmd = r"C:\Program Files\Tesseract-OCR\tesseract.exe"

router = APIRouter()

@router.post("/vision")
async def analizar_imagen(
    file: UploadFile = File(None),
    img_base64: str = Body(None)
):
    try:
        if file is not None:
            contenido = await file.read()
            if not contenido:
                raise HTTPException(status_code=400, detail="Archivo vacío.")
            imagen = Image.open(io.BytesIO(contenido))
            fuente = "archivo"
        elif img_base64 is not None:
            try:
                img_bytes = base64.b64decode(img_base64)
                imagen = Image.open(io.BytesIO(img_bytes))
                fuente = "base64"
            except Exception:
                raise HTTPException(status_code=400, detail="Error decodificando imagen base64.")
        else:
            raise HTTPException(status_code=400, detail="Debe subir una imagen (archivo o base64).")

        texto_extraido = pytesseract.image_to_string(imagen, lang="spa")
        if not texto_extraido.strip():
            raise HTTPException(status_code=422, detail="No se detectó texto legible en la imagen.")

        return {
            "texto_extraido": texto_extraido.strip(),
            "estatus": "OCR completado",
            "fuente": fuente
        }

    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Error analizando imagen: {str(e)}")
