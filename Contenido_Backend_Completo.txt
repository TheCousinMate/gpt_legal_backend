from fastapi import APIRouter, UploadFile, File, Request
from modules.reflexion import reflexionar
from modules.predict import procesar_prediccion
from modules.vision_forense import procesar_imagen
from modules.forensics_proxy import procesar_pdf
from modules.interprete import procesar_orden
import mimetypes

router = APIRouter()

@router.post("/universal")
async def router_universal(request: Request, file: UploadFile = File(None), mensaje: str = None):
    try:
        if file:
            tipo, _ = mimetypes.guess_type(file.filename.lower())
            if tipo:
                if tipo.startswith("image/"):
                    return await procesar_imagen(file)
                elif tipo == "application/pdf":
                    return await procesar_pdf(file)
                elif tipo == "application/json":
                    return await procesar_prediccion(file)
            return {"error": "Tipo de archivo no soportado."}

        elif mensaje:
            if mensaje.strip().lower().startswith("analiza") or "errores" in mensaje.lower():
                return reflexionar({"mensaje": mensaje})
            else:
                return await procesar_orden({"mensaje": mensaje})
        else:
            return {"error": "Debes proporcionar un mensaje o archivo para análisis."}
    except Exception as e:
        return {"error": f"Falla interna del sistema: {str(e)}"}

import requests
from bs4 import BeautifulSoup
from fastapi import APIRouter, HTTPException
from fastapi.responses import JSONResponse
# from utils_output import output_dual # Si lo necesitas, asegúrate de que utils_output.py exista

router = APIRouter()

@router.get("/api/bcn/norma/{id_norma}")
def norma_bcn(id_norma: int):
    """
    Obtiene los detalles de una norma de la Biblioteca del Congreso Nacional de Chile (BCN)
    mediante scraping.
    """
    try:
        url = f"https://www.bcn.cl/leychile/Navegar?idNorma={id_norma}"
        headers = {"User-Agent": "ForensicLegalBot/1.0"} # Buena práctica para identificar tu scraper
        response = requests.get(url, headers=headers, timeout=10) # Añadir timeout para evitar cuelgues
        response.raise_for_status() # Lanza una excepción para códigos de estado HTTP de error (4xx o 5xx)
        soup = BeautifulSoup(response.text, "html.parser")

        # Extracción del título
        titulo_tag = soup.find("h1") or soup.find("title") # Busca <h1> o <title>
        titulo = titulo_tag.get_text(strip=True) if titulo_tag else "Título no disponible"

        # Extracción de la fecha de publicación
        fecha_tag = soup.find(class_="fecha-publicacion")
        fecha = fecha_tag.get_text(strip=True) if fecha_tag else "Fecha no disponible"

        # Extracción del resumen o primer párrafo como contenido
        resumen_tag = soup.find("div", class_="resumen")
        if resumen_tag:
            contenido = resumen_tag.get_text(strip=True)
        else:
            primer_p = soup.find("p") # Intenta tomar el primer párrafo si no hay resumen
            contenido = primer_p.get_text(strip=True) if primer_p else "Contenido no disponible"

        resultado = {
            "norma": {
                "id": id_norma,
                "titulo": titulo,
                "fecha_publicacion": fecha,
                "contenido": contenido,
                "url_fuente": url
            },
            "estatus": "éxito",
            "fuente": "BCN - Ley Chile"
        }
        # Si usas output_dual, podrías hacer:
        # return JSONResponse(content=output_dual(resultado))
        return JSONResponse(content=resultado)

    except requests.exceptions.RequestException as e:
        raise HTTPException(status_code=503, detail=f"Error al conectar con BCN: {str(e)}. El servicio puede estar no disponible.")
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Error procesando norma BCN: {str(e)}")

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel # No se usa directamente aquí, pero es común
import requests
from bs4 import BeautifulSoup
from fastapi.responses import JSONResponse
# from utils_output import output_dual # Si lo necesitas, asegúrate de que utils_output.py exista

router = APIRouter()

@router.get("/legislacion/buscar")
def buscar_norma(palabra: str):
    """
    Busca normas en Ley Chile (BCN) por palabra clave y devuelve los primeros resultados.
    """
    try:
        # Codifica la palabra clave para la URL
        palabra_encoded = requests.utils.quote(palabra)
        url = f"https://www.bcn.cl/leychile/search?q={palabra_encoded}"
        
        headers = {"User-Agent": "ForensicLegalBot/1.0"}
        r = requests.get(url, headers=headers, timeout=15) # Añadir timeout
        r.raise_for_status() # Lanza HTTPError para respuestas 4xx/5xx
        soup = BeautifulSoup(r.text, 'html.parser')

        resultados = []
        # Seleccionar enlaces con la clase "resultado-titulo" que contienen el título de la norma
        enlaces = soup.select("h3.resultado-titulo a")

        for e in enlaces[:5]: # Limitar a los primeros 5 resultados
            # Asegúrate de que el 'href' es una URL completa
            norma_url = e['href'] if e['href'].startswith('http') else f"https://www.bcn.cl{e['href']}"
            resultados.append({
                "titulo": e.text.strip(),
                "url": norma_url
            })

        if not resultados:
            # Si no se encontraron resultados en el scraping
            return JSONResponse(content={
                "palabra": palabra,
                "resultados": [],
                "mensaje": "No se encontraron normas relevantes en BCN para la palabra clave.",
                "fuente": "https://www.bcn.cl/leychile"
            })

        return JSONResponse(content={
            "palabra": palabra,
            "resultados": resultados,
            "fuente": "https://www.bcn.cl/leychile"
        })

    except requests.exceptions.RequestException as e:
        raise HTTPException(status_code=503, detail=f"Error al conectar con BCN para búsqueda: {str(e)}. El servicio puede estar no disponible.")
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Error en el scraping de BCN: {str(e)}")
